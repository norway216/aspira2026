```cpp
// vector3d.h
#pragma once
#include <cmath>
#include <iostream>

namespace math {
template <typename T>
class Vec3 {
public:
    T x, y, z;
    Vec3(T x_ = 0, T y_ = 0, T z_ = 0) : x(x_), y(y_), z(z_) {}

    Vec3 operator+(const Vec3& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Vec3 operator*(T s) const { return {x*s, y*s, z*s}; }

    T dot(const Vec3& o) const { return x*o.x + y*o.y + z*o.z; }
    T length() const { return std::sqrt(x*x + y*y + z*z); }
    Vec3 normalized() const {
        T len = length();
        return len > 0 ? *this * (1/len) : *this;
    }
};

template <typename T>
std::ostream& operator<<(std::ostream& os, const Vec3<T>& v) {
    return os << "(" << v.x << ", " << v.y << ", " << v.z << ")";
}
} // namespace math
```

```cpp
// Engine.h — 头文件完全不暴露实现细节
#pragma once
#include <memory>
#include <vector>

class Engine {
public:
    Engine();
    ~Engine();                          // 必须在 .cpp 定义（Impl 是不完整类型）
    Engine(Engine&&) noexcept;          // 移动
    Engine& operator=(Engine&&) noexcept;
    Engine(const Engine&) = delete;     // 禁止拷贝

    bool load(const std::string& path);
    std::vector<float> infer(const std::vector<float>& input);

private:
    struct Impl;
    std::unique_ptr<Impl> m_impl;       // PIMPL 的核心
};

// Engine.cpp
#include <third_party/heavy_lib.h>      // 重量级第三方库只在这里 include

struct Engine::Impl {
    HeavyLib::Session session;
    std::vector<std::string> metadata;
    bool loaded = false;
};

Engine::Engine() : m_impl(std::make_unique<Impl>()) {}
Engine::~Engine() = default;            // 必须：Impl 在这里是完整类型
// ...
```

**PIMPL 的好处**：
- 第三方头文件不污染编译依赖
- 修改实现不需要重新编译所有 include Engine.h 的文件
- 本项目中的 `OnnxRuntimeEngine` 就是这样封装的

```cpp
class CameraSimulator {
    std::thread m_thread;
    std::atomic<bool> m_running{false};
    std::mutex m_mutex;
    cv::Mat m_latestFrame;

    void captureLoop() {
        int frameId = 0;
        while (m_running) {
            // 模拟采集一帧（实际会是 cv::VideoCapture::read）
            cv::Mat frame = generateTestPattern(frameId++);
            {
                std::lock_guard<std::mutex> lock(m_mutex);
                m_latestFrame = frame.clone();  // 在锁内拷贝
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(33)); // ~30 FPS
        }
    }

public:
    void start() {
        m_running = true;
        m_thread = std::thread(&CameraSimulator::captureLoop, this);
    }

    bool getFrame(cv::Mat& out) {
        std::lock_guard<std::mutex> lock(m_mutex);
        if (m_latestFrame.empty()) return false;
        out = m_latestFrame.clone();
        return true;
    }

    void stop() {
        m_running = false;
        if (m_thread.joinable()) m_thread.join();
    }
};
```

```cpp
cv::Mat preprocessEyeRoi(const cv::Mat& eyeRoi) {
    // 1. 转灰度
    cv::Mat gray;
    if (eyeRoi.channels() == 3)
        cv::cvtColor(eyeRoi, gray, cv::COLOR_BGR2GRAY);
    else
        gray = eyeRoi;

    // 2. CLAHE 对比度增强（比全局 equalizeHist 更适合虹膜）

    cv::Mat denoised;
    cv::medianBlur(enhanced, denoised, 3);

    return denoised;
}
```

### 第 9 周：边缘检测与霍夫变换

#### 必学内容

| 算法 | 原理 | 参数含义 |
|---|---|---|
| **Sobel** | 一阶导数近似 | dx, dy 方向，ksize |
| **Laplacian** | 二阶导数（旋转不变） | ksize，常用 3 |
| **Canny** | 多阶段：高斯平滑 → 梯度 → 非极大值抑制 → 双阈值连接 | low/high threshold |
| **HoughLines** | 直线：边缘点 → (ρ,θ) 参数空间投票 | rho, theta, threshold |
| **HoughCircles** | 圆：(x,y,r) 三维参数空间投票 | dp, minDist, param1, param2, minR, maxR |

#### 关键认知：HoughCircles 参数调优

```cpp
// dp — 累加器分辨率。最关键的性能参数！
//     dp=1: 累加器 = 原图尺寸（慢但精确）
//     dp=2: 累加器 = 原图/2（快 ~4×，精度下降）
// 虹膜检测：dp=1.2 是最佳平衡点

// param1 — Canny 高阈值（低阈 = 高阈/2）
//     100: 只保留强边缘
//     80: 多保留一些弱边缘

// param2 — 累加器阈值
//     30: 需要至少 30 个边缘点投票才能确认一个圆
//     20: 放宽要求，能检测到更多（可能误检）

// minDist — 圆心最小间距
//     设太小会检测到同心伪圆；设太大会漏掉相邻真圆

// minRadius/maxRadius — 半径范围
//     精确设置可以大幅减少假阳性
```

#### 本周练习：调参实验

```cpp
    auto t0 = std::chrono::steady_clock::now();
```

```cpp
cv::Mat myFilter2D(const cv::Mat& src, const cv::Mat& kernel) {
    int kh = kernel.rows / 2, kw = kernel.cols / 2;
    cv::Mat dst(src.size(), CV_32F, cv::Scalar(0));

    for (int y = kh; y < src.rows - kh; ++y) {
        for (int x = kw; x < src.cols - kw; ++x) {

            for (int ky = -kh; ky <= kh; ++ky)
                for (int kx = -kw; kx <= kw; ++kx)

                           kernel.at<float>(ky+kh, kx+kw);
            dst.at<float>(y, x) = sum;
        }
    }
    return dst;
}
```

```cpp
// 生成不同参数下的 Gabor 核，观察它们的纹理选择性
// θ=0   → 检测垂直纹理
// θ=π/2 → 检测水平纹理
// λ=8   → 波长长（粗纹理）
// λ=4   → 波长短（细纹理）

cv::Mat createGaborKernel(int size, float sigma, float theta,
                           float lambda, float gamma, float psi) {
    cv::Mat kernel(size, size, CV_32F);
    int half = size / 2;
    float sigmaX = sigma;
    float sigmaY = sigma / gamma;

    for (int y = -half; y <= half; ++y) {
        for (int x = -half; x <= half; ++x) {
            float xTheta = x*cos(theta) + y*sin(theta);
            float yTheta = -x*sin(theta) + y*cos(theta);

            float gauss = exp(-0.5 * (xTheta*xTheta/(sigmaX*sigmaX) +
                                       yTheta*yTheta/(sigmaY*sigmaY)));
            float sinusoid = cos(2*M_PI * xTheta/lambda + psi);

            kernel.at<float>(y+half, x+half) = gauss * sinusoid;
        }
    }
    // 去除直流分量（确保核总和为 0）
    cv::Scalar mean = cv::mean(kernel);
    kernel -= mean[0];
    return kernel;
}

cv::Mat display;
for (int i = 0; i < 8; ++i) {
    float theta = i * M_PI / 8;

    // 归一化到 [0, 255] 显示
    cv::normalize(kernel, kernel, 0, 255, cv::NORM_MINMAX);
```

### 第 13 周：Daugman 框架概述 + 虹膜分割

#### 理论内容

**Daugman 虹膜识别流水线（1993）：**

```
眼睛图像 → 虹膜定位 → 归一化(极坐标展开) → 特征编码(Gabor相位量化) → 匹配(汉明距离)
```

#### 虹膜分割的两种方法

| 方法 | 原理 | 优点 | 缺点 |
|---|---|---|---|
| **Hough 圆变换** | Canny 边缘 → 参数空间投票 | 简单、可解释 | 对噪声敏感，参数难调 |
| **微积分算子** (Daugman 原始) | `max(r,x₀,y₀) ‖∂/∂r ∮ I(x,y)/2πr ds‖` — 搜索最大化径向梯度的圆 | 精度高、不需要预处理 | 慢、需要初值 |

#### 本周练习：用线性搜索实现 Daugman 微积分算子

```cpp
// Daugman 微积分算子的离散版本
// 对于每个候选圆心 (cx, cy) 和半径 r：
//   沿圆周采样像素 → 计算径向梯度 → 找梯度最大的 (cx, cy, r)
// 这是比 HoughCircles 更符合虹膜生理特性的分割方法

float integroDifferential(const cv::Mat& gray, int cx, int cy, int r) {
    const int N = 128;  // 沿圆周采样点数
    float outer = 0, inner = 0;

    for (int i = 0; i < N; ++i) {
        float theta = 2 * M_PI * i / N;

        // 外圈 (r + dr)
        int xo = cx + (r+2) * cos(theta);
        int yo = cy + (r+2) * sin(theta);
        if (xo >= 0 && xo < gray.cols && yo >= 0 && yo < gray.rows)
            outer += gray.at<uint8_t>(yo, xo);

        // 内圈 (r - dr)
        int xi = cx + (r-2) * cos(theta);
        int yi = cy + (r-2) * sin(theta);
        if (xi >= 0 && xi < gray.cols && yi >= 0 && yi < gray.rows)
            inner += gray.at<uint8_t>(yi, xi);
    }

    return fabs(outer - inner) / (N * 4.0);
}
// 后续：在合理的圆心和半径范围内做粗搜索 → 精搜索
// 这是整个流水线中最"重"的计算，也是 GPU 加速的最优目标
```

### 第 14 周：Daugman 橡皮片归一化 + 掩码生成

#### 理论内容

**橡皮片模型**：将环形虹膜区域 [r_pupil, r_iris] × [0, 2π] 映射为矩形 [0, 63] × [0, 511]。

```
I(x(r,θ), y(r,θ)) → Î(r̂, θ̂)

其中：
  x(r,θ) = (1-r)·x_pupil(θ) + r·x_iris(θ)
  y(r,θ) = (1-r)·y_pupil(θ) + r·y_iris(θ)
  r ∈ [0,1] 是从瞳孔到虹膜边界的归一化半径
  θ ∈ [0, 2π] 是角度
```

#### 为什么用极坐标而非直角坐标？
- 虹膜是环形的，直角坐标展开会变形
- 极坐标展开后，瞳孔缩放可以在 r 方向补偿
- 后续的 Gabor 滤波在极坐标下做旋转补偿更容易

#### 本周练习：实现完整的 normalize + generateMasks

```cpp
// 输入：eyeRoi (256×256)，IrisBoundaries
// 输出：NormalizedIris (64×512 CV_32F)

NormalizedIris normalize(const cv::Mat& eyeRoi, const IrisBoundaries& b) {
    NormalizedIris result;
    result.image = cv::Mat(64, 512, CV_32F);
    result.mask  = cv::Mat(64, 512, CV_8U, cv::Scalar(0));

    for (int row = 0; row < 64; ++row) {
        float r = row / 63.0f;  // 0(瞳孔) → 1(虹膜)

        for (int col = 0; col < 512; ++col) {
            float theta = col * 2.0f * M_PI / 512.0f;

            // Daugman 线性插值：在瞳孔和虹膜边界之间插值
            float px = (1-r) * (b.pupil_center.x + b.pupil_radius*cos(theta))
                      +    r  * (b.iris_center.x  + b.iris_radius *cos(theta));
            float py = (1-r) * (b.pupil_center.y + b.pupil_radius*sin(theta))
                      +    r  * (b.iris_center.y  + b.iris_radius *sin(theta));

            // 双线性采样
            float val = sampleBilinear(eyeRoi, px, py);
            result.image.at<float>(row, col) = val;

            // 掩码：检查点是否在虹膜环内
            float distToIris = distance({px, py}, b.iris_center);
            float distToPupil = distance({px, py}, b.pupil_center);
            if (distToPupil >= b.pupil_radius*0.95f &&
                distToIris  <= b.iris_radius *1.02f) {
                result.mask.at<uint8_t>(row, col) = 255;
            }
        }
    }
    return result;
}
```

### 第 15 周：Gabor IrisCode 编码

#### 理论：为什么用相位而不是幅度？

```
对于 Gabor 核 h(r,θ) = exp(-r²/2σ²) · exp(j·2πf·r)
                                          ↑ 复指数 = cos + j·sin

卷积输出：O = I * h = |O| · exp(j·φ)
                       ↑幅度 ↑相位

相位 φ ∈ {0°, 90°, 180°, 270°} → 2 bits

幅度随光照、对比度变化剧烈 — 不可靠
相位只取决于纹理结构 — 可靠！
```

#### IrisCode 生成的完整流程

```
归一化虹膜 (64×512)
  │
  ├─ Gabor 滤波 (8方向 × 2相位 = 16个核)
  │    │
  │    └─ 对每个核的输出做 4×32 降采样
  │          ↓
  │    16 × 16 = 256 个采样点/方向
  │          ↓
  │    每个采样点 2 bits (实部符号 + 虚部符号)
  │          ↓
  │    8方向 × 256点 × 2bit = 4096 bits = 512 bytes
  │
  └─ 掩码位：对应位置的归一化掩码
```

#### 本周练习：实现完整的 extractIrisCode

```cpp
void extractIrisCode(const cv::Mat& normalizedIris,   // 64×512 CV_32F
                     const cv::Mat& mask,              // 64×512 CV_8U
                     std::vector<uint8_t>& irisCode,
                     std::vector<uint8_t>& maskCode) {

    int dsRow = 4, dsCol = 32;  // 降采样因子
    int codeRows = 16, codeCols = 16;
    size_t totalBits = codeRows * codeCols * 8 * 2;  // 8方向 × 2相位bit
    size_t totalBytes = (totalBits + 7) / 8;

    irisCode.assign(totalBytes, 0);
    maskCode.assign(totalBytes, 0);

    // 使用预计算好的 Gabor 核（不是每帧重建！）
    size_t bitIndex = 0;
    for (int orient = 0; orient < 8; ++orient) {
        cv::Mat responseEven, responseOdd;
        cv::filter2D(normalizedIris, responseEven, CV_32F, gaborCache[orient].even);
        cv::filter2D(normalizedIris, responseOdd,  CV_32F, gaborCache[orient].odd);

        for (int r = 0; r < codeRows; ++r) {
            for (int c = 0; c < codeCols; ++c) {
                int srcR = r * dsRow + dsRow/2;
                int srcC = c * dsCol + dsCol/2;

                float re = responseEven.at<float>(srcR, srcC);
                float im = responseOdd.at<float>(srcR, srcC);

                // 相位量化：实部符号 → bit0，虚部符号 → bit1
                uint8_t bit1 = (re > 0) ? 1 : 0;  // 0° vs 180°
                uint8_t bit2 = (im > 0) ? 1 : 0;  // 90° vs 270°

                irisCode[bitIndex/8] |= (bit1 << (bitIndex%8)); ++bitIndex;
                irisCode[bitIndex/8] |= (bit2 << (bitIndex%8)); ++bitIndex;
            }
        }
    }
}
```

### 第 16 周：汉明距离匹配 + 旋转补偿 + 活体检测

#### 理论内容

**分数汉明距离**（Fractional Hamming Distance）：

```
HD = ‖(codeA ⊕ codeB) ∩ maskA ∩ maskB‖ / ‖maskA ∩ maskB‖

⊕ = XOR（统计不同的 bit 数）
∩ mask = 只考虑虹膜掩码中标记为"有效"的位
```

**为什么是"分数"？** 因为不同虹膜图像的有效区域大小不同（睫毛遮挡等），需要用有效位数归一化。

**旋转补偿**：IrisCode 沿位方向循环移位 ±16 bits，取最小距离（因为虹膜可能在图像中旋转）。

#### 匹配阈值解读

| HD 值 | 含义 |
|---|---|
| HD ≈ 0.0 | 同一只眼睛（完美匹配） |
| HD ≤ 0.35 | 匹配阈值：判定为同一个人 |
| HD ≈ 0.5 | 不同人的虹膜（统计独立，期望值） |
| HD > 0.45 | 基本确定是不同人 |

#### 活体检测信号

| 信号 | 检测方法 | 攻击类型 |
|---|---|---|
| 纹理丰富度 | Laplacian 方差 | 打印照片（纹理丢失） |
| 高频能量 | Laplacian / FFT | 屏幕重放（莫尔纹） |
| 镜面反射 | 亮像素比例（理想 ~3%） | 美瞳、假眼 |
| 局部纹理一致性 | boxFilter 局部方差 | 二次翻拍 |
| 运动 | 帧间 absdiff | 静态图片攻击 |

#### 本周练习：实现完整的匹配+决策流水线

```cpp
struct MatchResult {
    std::string user_id;
    float hamming_dist;
    float similarity;  // 1.0 - hamming_dist
    bool matched;      // hamming_dist <= threshold
};

MatchResult search(const IrisEmbedding& query,
                   const std::vector<IrisTemplate>& templates) {
    MatchResult best;
    best.hamming_dist = 1.0f;

    for (const auto& tmpl : templates) {
        // 33 个旋转偏移 (-16...+16)
        for (int shift = -16; shift <= 16; ++shift) {
            float hd = hammingDistance(
                query.iris_code, query.mask_code,
                tmpl.iris_code, tmpl.mask_code, shift);

            if (hd < best.hamming_dist) {
                best.hamming_dist = hd;
                best.user_id = tmpl.user_id;
            }
        }
    }

    best.similarity = 1.0f - best.hamming_dist;
    best.matched = (best.hamming_dist <= 0.35f);
    return best;
}
```

### 第 17 周：神经网络基础 + YOLO 原理

#### 理论内容

| 主题 | 核心概念 |
|---|---|
| 卷积神经网络 (CNN) | 卷积层、池化层、全连接层、激活函数 |
| 目标检测基础 | 分类 vs 定位 vs 检测；bounding box 回归 |
| YOLO 的核心思想 | 一次前向传播同时做分类和定位；将图像划分为 S×S 网格 |
| 锚框 (Anchor) | 预设的宽高比先验框，YOLOv5 中用 3 种大小 × 3 个检测头 = 9 个锚框 |
| 无锚框 (Anchor-free) | YOLOv8+ 直接回归中心点 + 宽高 |
| 交并比 (IoU) | IoU = A∩B / A∪B；衡量框重叠程度 |
| NMS | 贪心算法：按置信度排序 → 逐个排除 IoU 过高的重叠框 |

#### YOLO 输出格式差异

```
YOLOv5 (anchor-based, 3 outputs):
  Output 0: [1, (5+nc)×3, 80, 80]    ← P3/8 小目标
  Output 1: [1, (5+nc)×3, 40, 40]    ← P4/16 中目标
  Output 2: [1, (5+nc)×3, 20, 20]    ← P5/32 大目标
  每个通道 = [tx, ty, tw, th, obj_conf, cls0...cls79]
  需要 sigmoid(tx) + anchor 来解码真实坐标

YOLOv8 (anchor-free, 1 output):
  Output: [1, 4+nc, 8400]             ← 所有网格拼接
  8400 = 80×80 + 40×40 + 20×20
  字段 = [cx, cy, w, h, cls0...cls79]
  坐标直接是相对于输入尺寸的，不需要 anchor 解码
  内存布局：channel-major [fields, proposals]
```

#### 本周练习：YOLOv8 输出解码

```cpp
// 输入：ONNX Runtime 返回的 flat float vector (shape [1, 84, 8400])
// 需要理解 channel-major 的跨步访问方式

std::vector<Detection> decodeYoloOutput(
    const std::vector<float>& output,  // size = 84 * 8400
    int numProposals, int numClasses,
    float confThreshold)
{
    std::vector<Detection> detections;
    int fields = 4 + numClasses;  // cx, cy, w, h, cls0...

    for (int i = 0; i < numProposals; ++i) {
        // Channel-major 布局：output[field * numProposals + proposal_index]
        float cx = output[0 * numProposals + i];  // 不是 output[i*fields+0]！
        float cy = output[1 * numProposals + i];
        float w  = output[2 * numProposals + i];
        float h  = output[3 * numProposals + i];

        // 找最大类别置信度
        float maxConf = 0;
        int bestClass = 0;
        for (int c = 0; c < numClasses; ++c) {
            float conf = output[(4 + c) * numProposals + i];
            if (conf > maxConf) { maxConf = conf; bestClass = c; }
        }

        if (maxConf >= confThreshold) {
            detections.push_back({cx, cy, w, h, maxConf, bestClass});
        }
    }
    return detections;
}
// 关键理解：
// output 在内存中不是 [proposal0_fields, proposal1_fields, ...]
// 而是 [all_cx, all_cy, all_w, all_h, all_cls0, all_cls1, ...]
// 这意味着遍历 proposal 时需要跨步访问 numProposals
```

### 第 19 周：模型预处理 — YOLO 输入流水线

#### 理论：为什么需要 Letterbox 而不是直接 resize？

```
直接 resize 640×480 → 640×640：
┌──────────┐     ┌──────────────┐
│  🧑      │     │    🧑       │  ← 人脸被压扁了
│          │ ──→ │              │     宽高比不对！
└──────────┘     └──────────────┘

Letterbox 后：
┌──────────┐     ┌──────────────┐
│  🧑      │     │░░░🧑░░░░░░░│  ← 保持宽高比
│          │ ──→ │░░░░░░░░░░░░│     灰色填充
└──────────┘     └──────────────┘
```

#### 完整的 YOLO 预处理流程

```cpp
// 步骤 1-5 必须在 C++ 中手工实现（OpenCV 的 blobFromImage 不够灵活）
PreprocessResult preprocess(const cv::Mat& frame, int inputSize) {
    // 1. 计算缩放比例和 letterbox padding
    float scale = std::min(inputSize / (float)frame.cols,
                            inputSize / (float)frame.rows);
    int scaledW = frame.cols * scale;
    int scaledH = frame.rows * scale;
    int padX = (inputSize - scaledW) / 2;
    int padY = (inputSize - scaledH) / 2;

    // 2. 缩放
    cv::Mat resized;
    cv::resize(frame, resized, cv::Size(scaledW, scaledH),
               0, 0, cv::INTER_LINEAR);

    // 3. Letterbox 填充 (YOLO 用 114,114,114 灰色)
    cv::Mat letterbox(inputSize, inputSize, CV_8UC3,
                       cv::Scalar(114, 114, 114));
    resized.copyTo(letterbox(cv::Rect(padX, padY, scaledW, scaledH)));

    // 4. BGR → RGB
    cv::Mat rgb;
    cv::cvtColor(letterbox, rgb, cv::COLOR_BGR2RGB);

    // 5. uint8 [0,255] → float [0,1]
    rgb.convertTo(rgb, CV_32F, 1.0 / 255.0);

    // 6. HWC → CHW (最易出错的步骤！)
    std::vector<float> blob(3 * inputSize * inputSize);
    for (int h = 0; h < inputSize; ++h) {
        const float* row = rgb.ptr<float>(h);
        for (int w = 0; w < inputSize; ++w) {
            size_t idx = h * inputSize + w;
            blob[0 * inputSize*inputSize + idx] = row[w*3 + 0];  // R
            blob[1 * inputSize*inputSize + idx] = row[w*3 + 1];  // G
            blob[2 * inputSize*inputSize + idx] = row[w*3 + 2];  // B
        }
    }

    return {blob, scale, scale, (float)padX, (float)padY};
}
```

### 第 24 周：SIMD 优化 — 永远用 OpenCV 函数

#### 理论：为什么 OpenCV 比你手写的循环快？


### 第 25 周：预计算与缓存 — 消除重复计算

#### 识别模式：输入相关 vs 输入无关

```
每个函数调用的参数分两类：

输入相关（每帧不同）：像素值、图像尺寸、光照条件
输入无关（每帧相同）：Gabor 核参数、高斯核 sigma、形态学核形状

优化策略：
  输入无关 → 在 initialize() 中预计算 → 存入缓存 → 每帧直接用

```

|---|---|---|---|---|
| FFT (莫尔纹检测) | O(N log N) ~5ms | Laplacian 方差 | O(N) ~0.05ms | 两者都是高频能量指标；屏幕重放导致异常高频 |
| LBP (8邻域+熵) | O(9N) ~2.5ms | boxFilter 方差 | O(N) ~0.1ms | 两者衡量局部纹理多样性；假眼缺乏自然纹理变化 |
